% Shared preamble for every figure. Each figure is  \documentclass{standalone}  +  \input{figstyle}.

% Sizing. The paper is arXiv single column: article 10 pt, 1 in margins on US letter, so \textwidth = 6.5 in.
% Figures are drawn at their final size and included in the paper WITHOUT scaling, so 8 pt here is 8 pt on the page

% ---- colour: one job per colour, never colour alone ---------------------------------------------------------
% Okabe-Ito hues (validated: lightness band, chroma, CVD separation). Every series also has its own marker and
% line style, so the figures read in greyscale and to colour-vision-deficient readers (make gray renders them).
\definecolor{cFV}{HTML}{0072B2}     % finite-volume inverse (hard physics)
\definecolor{cPINN}{HTML}{D55E00}   % physics-informed neural network (soft physics)
\definecolor{cAlt}{HTML}{009E73}    % third series (aldFoam case C, second dose ...)
\definecolor{cAlt2}{HTML}{CC79A7}   % fourth series
\definecolor{cObs}{HTML}{000000}    % measured data: always black
\definecolor{cInk2}{HTML}{52514E}   % secondary text
\definecolor{cMuted}{HTML}{898781}  % muted ink, excluded points
\definecolor{cGrid}{HTML}{E1E0D9}
\definecolor{cBand}{HTML}{0072B2}   % uncertainty bands: series colour at low opacity
\definecolor{cFail}{HTML}{C00000}
% deposition temperature (a magnitude): one hue, light to dark, plus a marker per temperature (matplotlib Purples)
\definecolor{t100}{rgb}{0.6178,0.6021,0.7835}
\definecolor{t125}{rgb}{0.5070,0.4951,0.7318}
\definecolor{t150}{rgb}{0.4218,0.3298,0.6456}
\definecolor{t175}{rgb}{0.3325,0.1588,0.5636}
\definecolor{t200}{rgb}{0.2471,0.0000,0.4902}
\pgfplotsset{
  T100/.style={t100, mark=*, mark size=1.3pt},
  T125/.style={t125, mark=square*, mark size=1.2pt},
  T150/.style={t150, mark=triangle*, mark size=1.7pt},
  T175/.style={t175, mark=diamond*, mark size=1.7pt},
  T200/.style={t200, mark=pentagon*, mark size=1.6pt},
  % keep only rows where column #1 equals #2 (other rows are discarded, not drawn)
  only/.style 2 args={unbounded coords=discard, x filter/.expression={\thisrow{#1}==#2 ? x : nan}},
}

  sPINN/.style={cPINN, line width=0.9pt, dashed, mark=triangle*, mark size=1.8pt, mark options={solid, fill=cPINN}},
  sObs/.style={only marks, mark=*, mark size=1.3pt, cObs, mark options={fill=white, line width=0.5pt}},
  sObsX/.style={only marks, mark=o, mark size=1.3pt, cMuted},
  sGuide/.style={cMuted, line width=0.4pt, densely dotted},

% ---- checks: a figure whose check failed is stamped, never quietly polished --------------------------------------
% data/checks.tex is written by code/export_figdata.py: \figcheckstatus{fig3}{PASS} etc.
\makeatletter
\newcommand{\figcheckstatus}[2]{\expandafter\gdef\csname figcheck@#1\endcsname{#2}}
\InputIfFileExists{data/checks.tex}{}{\typeout{figstyle: data/checks.tex missing: run make data}}
% \figcheck{fig3}: place inside the tikzpicture, after everything else.
\newcommand{\figcheck}[1]{%
  \@ifundefined{figcheck@#1}{\def\@fcs{MISSING}}{\edef\@fcs{\csname figcheck@#1\endcsname}}%
  \def\@pass{PASS}%
  \ifx\@fcs\@pass\else
    \node[draw=cFail, line width=1.2pt, text=cFail, fill=white, fill opacity=0.85, text opacity=1,
          font=\small\bfseries, align=center, rotate=12]
      at (current bounding box.center) {CHECK #1: \@fcs\\[-1pt]\scriptsize see paper/figures/data/checks.csv};
  \fi}
\makeatother
