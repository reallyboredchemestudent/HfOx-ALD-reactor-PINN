  \node[draw=cInk2, line width=0.4pt, rounded corners=1.5pt, inner sep=3.5pt, anchor=north, align=center] (eq) at (1.0,-0.48)

     $\cov = 1 - \mathrm{e}^{-\expo}$, \quad inlet flux $\propto \reach/\zb$};
  \node[anchor=north, font=\scriptsize, align=center, text=cInk2] at (eq.south)
